function postedState(u) {

  if (!u.lastPosted) return { level:'never', label:'ยังไม่โพสต์', cls:'text-amber-600' };

  const d = new Date(u.lastPosted);

  const timeStr = d.toLocaleTimeString('th-TH', { hour: '2-digit', minute: '2-digit' });

  const today = new Date(); today.setHours(0,0,0,0);

  const postDay = new Date(u.lastPosted); postDay.setHours(0,0,0,0);

  const days = Math.floor((today - postDay) / 86400000);

  const cnt = u.postCount ? ` · ${u.postCount} ครั้ง` : '';

  const time = ` · ล่าสุด ${timeStr}`;

  if (days === 0) return { level:'fresh', label:`โพสต์วันนี้${cnt}${time}`, cls:'text-emerald-600' };

  if (days === 1) return { level:'ok', label:`โพสต์เมื่อวาน${cnt}${time}`, cls:'text-slate-500' };

  if (days < STALE_DAYS) return { level:'aging', label:`โพสต์ ${days} วันก่อน${cnt}${time}`, cls:'text-slate-500' };

  return { level:'stale', label:`⚠ โพสต์ ${days} วันก่อน${cnt}${time}`, cls:'text-amber-600 font-semibold' };

}

function excelCellToDate(v) {

  if (v == null || v === '') return '';

  if (v instanceof Date) return v.toISOString().slice(0, 10);

  const s = String(v).trim();

  const m = s.match(/^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/);

  if (m) return `${m[1]}-${String(m[2]).padStart(2, '0')}-${String(m[3]).padStart(2, '0')}`;

  return s;

}

function setWm(v) { wmOn = v; try { localStorage.setItem(WM_KEY, v ? '1' : '0'); } catch (e) {} toast(v ? 'จะใส่ลายน้ำโลโก้ในรูปใหม่' : 'ปิดลายน้ำแล้ว'); }

function loadLogo() {

  if (logoTried) return Promise.resolve(logoImg);

  logoTried = true;

  return new Promise(res => { const im = new Image(); im.onload = () => { logoImg = im; res(im); }; im.onerror = () => res(null); im.src = 'logo.png'; });

}

function drawWatermark(ctx, w, h) {

  if (!logoImg) return;

  const lw = Math.round(w * 0.16), lh = Math.round(lw * logoImg.height / logoImg.width), pad = Math.round(w * 0.03);

  ctx.save();

  ctx.globalAlpha = 0.9;

  ctx.shadowColor = 'rgba(255,255,255,0.9)'; ctx.shadowBlur = Math.round(lw * 0.08);

  ctx.drawImage(logoImg, w - lw - pad, pad, lw, lh);

  ctx.restore();

}

function decodeImage(file) {

  return new Promise((res, rej) => {

    const img = new Image(); const url = URL.createObjectURL(file);

    img.onload = () => { URL.revokeObjectURL(url); res(img); };

    img.onerror = () => { URL.revokeObjectURL(url); rej(new Error('อ่านรูปไม่ได้')); };

    img.src = url;

  });

}

function toJpeg(c, q) { return new Promise((res, rej) => c.toBlob(b => b ? res(b) : rej(new Error('encode failed')), 'image/jpeg', q)); }

function scaleCanvas(img, max) {

  const r = Math.min(1, max / Math.max(img.width, img.height));

  const c = document.createElement('canvas'); c.width = Math.round(img.width * r); c.height = Math.round(img.height * r);

  c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);

  return c;

}

async function processPhoto(file) {

  const img = await decodeImage(file);

  const fullC = scaleCanvas(img, 1600);

  if (wmOn) { await loadLogo(); drawWatermark(fullC.getContext('2d'), fullC.width, fullC.height); }

  const thumbC = document.createElement('canvas');

  const tr = Math.min(1, 400 / Math.max(fullC.width, fullC.height));

  thumbC.width = Math.round(fullC.width * tr); thumbC.height = Math.round(fullC.height * tr);

  thumbC.getContext('2d').drawImage(fullC, 0, 0, thumbC.width, thumbC.height);

  return [await toJpeg(fullC, 0.82), await toJpeg(thumbC, 0.7)];

}

async function makeCollage(selectedPhotos) {

  const u = units.find(x => x.id === galleryUnitId); if (!u || !selectedPhotos.length) return;

  const prog = document.getElementById('galProgress'); prog.classList.remove('hidden'); prog.textContent = 'กำลังสร้างภาพรวม...';

  try {

    await loadLogo();

    const pics = selectedPhotos.slice(0, 6);

    const W = 1080, cols = 2, gap = 14, pad = 36, headH = 250, footH = 128;

    const cell = (W - pad * 2 - gap * (cols - 1)) / cols;

    const rows = Math.ceil(pics.length / cols);

    const H = headH + rows * cell + (rows - 1) * gap + pad + footH;

    const c = document.createElement('canvas'); c.width = W; c.height = H;

    const ctx = c.getContext('2d');

    const Fn = 'Inter, "Noto Sans Thai", -apple-system, sans-serif';

    ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, W, H);

    ctx.textAlign = 'left'; ctx.textBaseline = 'alphabetic';

    let textMax = W - pad * 2;

    if (logoImg) { const lh = 150, lw = lh * logoImg.width / logoImg.height; ctx.drawImage(logoImg, W - pad - lw, 40, lw, lh); textMax = W - pad * 2 - lw - 30; }

    const fit = (text, font) => { ctx.font = font; while (ctx.measureText(text).width > textMax && text.length > 3) text = text.slice(0, -2).trimEnd() + '…'; return text; };

    ctx.fillStyle = '#1c1917'; ctx.font = `bold 52px ${Fn}`;

    ctx.fillText(fit(u.project, `bold 52px ${Fn}`), pad, 96);

    const specs = [bedLabelTH(u.type), u.bath ? `${u.bath} ห้องน้ำ` : '', u.size ? `${u.size} ตร.ม.` : '', u.floor ? `ชั้น ${u.floor}` : ''].filter(Boolean).join('  ·  ');

    ctx.fillStyle = '#6b6058'; ctx.font = `34px ${Fn}`;

    ctx.fillText(fit(specs, `34px ${Fn}`), pad, 152);

    ctx.fillStyle = '#b85c38'; ctx.font = `bold 46px ${Fn}`;

    if (hasActivePromo(u)) ctx.fillText(`🔥 ฿${fmtNum(u.promoRent)} (จาก ฿${fmtNum(u.rent)})`, pad, 212);

    else ctx.fillText(`฿${fmtNum(u.rent)} / เดือน`, pad, 212);

    for (let i = 0; i < pics.length; i++) {

      const img = await decodeImage(pics[i].full);

      const x = pad + (i % cols) * (cell + gap), y = headH + Math.floor(i / cols) * (cell + gap);

      const sz = Math.min(img.width, img.height), sx = (img.width - sz) / 2, sy = (img.height - sz) / 2;

      ctx.save(); ctx.beginPath(); ctx.roundRect(x, y, cell, cell, 20); ctx.clip();

      ctx.drawImage(img, sx, sy, sz, sz, x, y, cell, cell); ctx.restore();

    }

    ctx.fillStyle = '#b85c38'; ctx.fillRect(0, H - footH, W, footH);

    ctx.fillStyle = '#ffffff'; ctx.font = `bold 32px ${Fn}`;

    const cfg = getPostConfig();

    ctx.fillText(cfg.agentName + ' Real Estate Agent', pad, H - footH + 48);

    ctx.font = `26px ${Fn}`; ctx.fillStyle = '#f5d9cc';

    ctx.fillText(`LINE: ${cfg.lineId}   ·   โทร ${cfg.phone}`, pad, H - footH + 92);

    const blob = await toJpeg(c, 0.88);

    const file = new File([blob], `${u.project.replace(/\s+/g, '_')}_${u.room}_overview.jpg`, { type: 'image/jpeg' });

    prog.classList.add('hidden');

    if (navigator.canShare && navigator.canShare({ files: [file] })) {

      try { await navigator.share({ files: [file] }); } catch (e) { if (e.name !== 'AbortError') toast('แชร์ไม่สำเร็จ'); }

    } else { const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = file.name; a.click(); }

  } catch (e) { prog.classList.add('hidden'); toast('สร้างภาพรวมไม่สำเร็จ: ' + e.message); }

}

let galleryUnitId = null, galleryPhotos = [], lbIndex = 0;

const photoUrlCache = new Map();

function thumbUrlOf(p) { let e = photoUrlCache.get(p.id); if (!e) { e = {}; photoUrlCache.set(p.id, e); } if (!e.thumbUrl) e.thumbUrl = URL.createObjectURL(p.thumb); return e.thumbUrl; }

function fullUrlOf(p) { let e = photoUrlCache.get(p.id); if (!e) { e = {}; photoUrlCache.set(p.id, e); } if (!e.fullUrl) e.fullUrl = URL.createObjectURL(p.full); return e.fullUrl; }

function clearPhotoUrlCache() { photoUrlCache.forEach(e => { if (e.thumbUrl) URL.revokeObjectURL(e.thumbUrl); if (e.fullUrl) URL.revokeObjectURL(e.fullUrl); }); photoUrlCache.clear(); }

function pickPhotos(src) { document.getElementById(src === 'camera' ? 'fileCamera' : 'fileAlbum').click(); }

async function importPhotos(files) {

  if (!files || !files.length || !galleryUnitId) return;

  const prog = document.getElementById('galProgress'); prog.classList.remove('hidden');

  let ok = 0; const base = Date.now();

  for (let i = 0; i < files.length; i++) {

    prog.textContent = `กำลังบันทึกรูป ${i + 1}/${files.length}...`;

    try {

      const [full, thumb] = await processPhoto(files[i]);

      await req(tx('readwrite').put({ id: 'p' + base + '_' + i, unitId: galleryUnitId, full, thumb, ts: base + i, order: base + i }));

    } catch (e) {}

  }

  prog.classList.add('hidden');

  document.getElementById('fileAlbum').value = ''; document.getElementById('fileCamera').value = '';

  await refreshCounts(); await renderGallery(); render();

  toast(ok === files.length ? `เพิ่มรูป ${ok} รูปแล้ว` : `บันทึกได้ ${ok}/${files.length} รูป`);

}

  pickerMode = mode;

  pickerSelected = [];

  document.getElementById('pickTitle').textContent = mode === 'collage' ? 'เลือกรูปสำหรับภาพรวม' : 'เลือกรูปที่จะส่ง';

    : 'แตะเพื่อเลือก/ยกเลิก · ลำดับที่เลือกจะแสดงบนรูป';

  document.getElementById('pickConfirmLabel').textContent = mode === 'collage' ? 'สร้างภาพรวม' : 'ส่งรูป';

  renderPicker();

  document.getElementById('pickOverlay').classList.remove('hidden');

  document.getElementById('pickSheet').classList.add('open');

  lucide.createIcons();

}

function closePhotoPicker() {

  document.getElementById('pickOverlay').classList.add('hidden');

  document.getElementById('pickSheet').classList.remove('open');

  pickerMode = null; pickerSelected = [];

}

function renderPicker() {

  const grid = document.getElementById('pickGrid');

  grid.innerHTML = galleryPhotos.map(p => {

    const idx = pickerSelected.indexOf(p.id);

    const isSel = idx >= 0;

    return `<div class="pick-cell relative aspect-square rounded-lg overflow-hidden bg-slate-800 ${isSel ? 'selected' : ''}" onclick="togglePick('${p.id}')">

      <img src="${thumbUrlOf(p)}" class="w-full h-full object-cover pointer-events-none" draggable="false" alt="">

      ${isSel ? `<div class="order-badge">${idx + 1}</div>` : ''}

    </div>`;

  }).join('');

  const c = pickerSelected.length;

  document.getElementById('pickCount').textContent = `เลือกแล้ว ${c} รูป${pickerMode === 'collage' ? ' / สูงสุด 6' : ''}`;

  document.getElementById('pickConfirm').disabled = c === 0;

}

function togglePick(id) {

  const i = pickerSelected.indexOf(id);

  if (i >= 0) pickerSelected.splice(i, 1);

  else {

    if (pickerMode === 'collage' && pickerSelected.length >= 6) return toast('เลือกได้สูงสุด 6 รูป');

    pickerSelected.push(id);

  }

  renderPicker();

}

function pickAll() {

  if (pickerMode === 'collage') pickerSelected = galleryPhotos.slice(0, 6).map(p => p.id);

  else pickerSelected = galleryPhotos.map(p => p.id);

  renderPicker();

}

function pickNone() { pickerSelected = []; renderPicker(); }

function confirmPhotoPicker() {

  if (!pickerSelected.length) return;

  const picked = pickerSelected.map(id => galleryPhotos.find(p => p.id === id)).filter(Boolean);

  const mode = pickerMode;

  closePhotoPicker();

  if (mode === 'collage') makeCollage(picked);

  else shareSelectedPhotos(picked);

}

async function shareSelectedPhotos(photos) {

  const u = units.find(x => x.id === galleryUnitId); if (!u) return;

  const files = photos.map((p, i) => new File([p.full], `${u.project.replace(/\s+/g, '_')}_${u.room}_${i + 1}.jpg`, { type: 'image/jpeg' }));

  if (navigator.canShare && navigator.canShare({ files })) {

    try { await navigator.share({ files }); } catch (e) { if (e.name !== 'AbortError') toast('แชร์ไม่สำเร็จ'); }

  } else {

    files.forEach(f => { const a = document.createElement('a'); a.href = URL.createObjectURL(f); a.download = f.name; a.click(); });

    toast('เบราว์เซอร์นี้แชร์รูปตรงไม่ได้ ดาวน์โหลดให้แทนแล้ว');

  }

}
